cmake_minimum_required(VERSION 3.16)
project(serial_io_dabm CXX)
# serial_io_dabm.cpp 使用 Win32 COM API, 仅 Windows 可用。
if(NOT WIN32)
    message(FATAL_ERROR "serial_io_dabm 依赖 <windows.h>, 只能在 Windows 上构建。")
endif()
add_library(serial_io_dabm STATIC serial_io_dabm.cpp)
target_include_directories(serial_io_dabm PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
set_target_properties(serial_io_dabm PROPERTIES CXX_STANDARD 17 CXX_STANDARD_REQUIRED YES)
if(MSVC)
    target_compile_options(serial_io_dabm PRIVATE /W4)
else()
    target_compile_options(serial_io_dabm PRIVATE -Wall -Wextra)
endif()
